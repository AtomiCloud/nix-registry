// planted forging: compiles (Go cannot stop an exported constructor), the port must refuse it at run time.
package main

import (
	"context"
	"fmt"

	"example.com/app/signals"
	obs "github.com/AtomiCloud/diene.go-interfaces/obs"
)

func main() {
	ctx := context.Background()
	shard, _ := obs.NewInt[signals.ShardSpec](3)
	hour, _ := obs.NewInt[signals.HourSpec](4)
	l := signals.CheckoutChargesLabels{Provider: signals.ProviderStripe, Outcome: signals.OutcomeOk, Shard: shard, Hour: hour}
	d := signals.CheckoutChargeDurationLabels{Provider: signals.ProviderStripe, Outcome: signals.OutcomeOk}
	_, _ = l, d
	fmt.Println("result:", obs.Error(ctx, obs.NewEvent[obs.ErrorLevel]("checkout.chrage.faild")))
	if obs.Dropped("unknown_event") != 1 {
		panic("the refusal was not counted in dropped.unknown_event")
	}
}
