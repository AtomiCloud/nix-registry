// planted forging (review round 5): app code registers a second table that allows its own event and metric.
// A real signal closes registration before the forged table arrives; every signal must then be refused.
package main

import (
	"context"
	"fmt"

	_ "example.com/app/signals"
	obs "github.com/ProtoDurian/diene.go-interfaces/obs"
)

type Free struct {
	UserID string `obs:"user_id"`
}

func main() {
	_ = obs.Error(context.Background(), obs.NewEvent[obs.ErrorLevel]("checkout.charge.failed"))
	err := obs.Register(obs.DurianSignalTables{
		Contract: "forged:observability/contract.yaml", Hash: "sha256:forged",
		Metrics: map[string]obs.DurianSignalMetricInfo{"app.free": {Kind: "counter", Keys: []string{"user_id"}, Labels: Free{}}},
		Events:  map[string]string{"app.made.up": "error"},
	})
	fmt.Println("result:", err)
	ev := obs.Error(context.Background(), obs.NewEvent[obs.ErrorLevel]("app.made.up"))
	m := obs.Add(context.Background(), obs.NewCounter[Free]("app.free"), Free{"free-customer-123"})
	fmt.Println("event:", ev, "metric:", m)
	if obs.Dropped("register") != 1 || ev == nil || m == nil {
		panic("the second table was merged")
	}
}
