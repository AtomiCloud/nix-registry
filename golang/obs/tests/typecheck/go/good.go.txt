package main

import (
	"fmt"
	"time"

	"context"
	"example.com/app/signals"
	obs "github.com/AtomiCloud/diene.go-interfaces/obs"
)

func main() {
	shard, err := obs.NewInt[signals.ShardSpec](3)
	if err != nil {
		panic(err)
	}
	if _, err := obs.NewInt[signals.ShardSpec](16); err == nil {
		panic("16 accepted")
	}
	hour, err := obs.AtTime[signals.HourSpec](time.Now())
	if err != nil {
		panic(err)
	}
	l := signals.CheckoutChargesLabels{Provider: signals.ProviderStripe, Outcome: signals.OutcomeOk, Shard: shard, Hour: hour}
	var _ obs.DurianSignalCounter[signals.CheckoutChargesLabels] = signals.CheckoutCharges
	var _ obs.DurianSignalHistogram[signals.CheckoutChargeDurationLabels] = signals.CheckoutChargeDuration
	var _ obs.Event[obs.ErrorLevel] = signals.ChargeFailed
	zero, _ := obs.NewInt[signals.ShardSpec](0)
	if !zero.Valid() || (signals.Shard{}).Valid() || (signals.Provider{}).Valid() || !signals.ProviderStripe.Valid() {
		panic("a valid 0 must differ from an unset value")
	}
	if err := obs.Add(context.Background(), signals.CheckoutCharges, l); err != nil {
		panic(err)
	}
	if err := obs.Error(context.Background(), signals.ChargeFailed); err != nil {
		panic(err)
	}
	bad := l
	bad.Shard = signals.Shard{}
	if obs.Add(context.Background(), signals.CheckoutCharges, bad) == nil {
		panic("unset bounded accepted")
	}
	bad = l
	bad.Provider = signals.Provider{}
	if obs.Add(context.Background(), signals.CheckoutCharges, bad) == nil {
		panic("zero enum accepted")
	}
	if err := obs.Record(context.Background(), signals.CheckoutChargeDuration, signals.CheckoutChargeDurationLabels{Provider: signals.ProviderAdyen, Outcome: signals.OutcomeOk}); err != nil {
		panic(err)
	}
	// The generated init() registered the tables: every valid call above passed, nothing was dropped
	// for an unknown name, level or key (the two planted unset labels count as unset_label).
	for _, r := range []string{"unknown_event", "unknown_metric", "unknown_key"} {
		if obs.Dropped(r) != 0 {
			panic("a valid generated call was dropped as " + r)
		}
	}
	if obs.Dropped("unset_label") != 2 {
		panic("unset labels not counted")
	}
	fmt.Println(l.Provider, signals.ChargeFailed.Name(), signals.ChargeFailed.Level(), signals.MetricTable["checkout.charges"].Keys)
}
